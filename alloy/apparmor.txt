#include <tunables/global>

# Profile for the Grafana Alloy add-on. Built from the Home Assistant template
# (developers.home-assistant.io/docs/apps/presentation#apparmor) plus what the
# two s6 scripts and the Alloy binary actually touch.
#
# ENFORCED, and tested: CI loads this profile on the runner, runs the freshly
# built image under it against a real journal and a real Loki, and fails the
# build on any apparmor DENIED for it (.github/scripts/e2e.sh). A dependency
# bump that needs a new path shows up there, not inside a household's Home
# Assistant. On a real host, denials are `journalctl _TRANSPORT="audit"` lines;
# with the journal shipped to Loki they are also queryable there.
profile alloy flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # No blanket `file,` rule (the Home Assistant template has one, and with it
  # every path rule below would be decoration). File access is what is listed
  # here and nothing else; the list was taken from a complain-mode run of the
  # whole e2e, every option path included, and the gate keeps it honest.
  signal (send) set=(kill,term,int,hup,cont),

  # Docker stops the container through runc. Where runc runs under an AppArmor
  # profile of its own (AppArmor 4 hosts such as the Ubuntu 24.04 runner), its
  # SIGTERM/SIGKILL arrive from peer "runc" and are denied without this rule;
  # Docker's own default profile carries the same rule (moby/moby#47749).
  # AppArmor 3.1.7, which Home Assistant OS ships, has no runc profile, so
  # there runc is unconfined and the rule is never used.
  signal (receive) peer=runc,

  # S6-Overlay (v3 layout: s6-rc.d under /etc/s6-overlay). Its stages are
  # scripts, so they are read as well as executed.
  /init rix,
  /package/** r,
  /etc/fix-attrs.d/ r,
  /etc/services.d/ r,

  # Users and groups: s6-overlay-suexec, bash and the HEALTHCHECK's curl
  /etc/passwd r,
  /etc/group r,
  /etc/nsswitch.conf r,

  # The HEALTHCHECK's curl loads OpenSSL's config. AppArmor 4 allows that in
  # abstractions/base; the 3.1.7 Home Assistant OS ships does not.
  /etc/ssl/openssl.cnf r,

  # The variant marker the generator reads (alloy-host has one, alloy not)
  /etc/alloy-addon/variant r,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rix,
  # s6-overlay's legacy stages list and run these; nothing writes there
  # (the template's rwix allowed writing a file and then executing it).
  /etc/services.d/** rix,
  /etc/cont-init.d/** rix,
  /etc/cont-finish.d/** rix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Scratch space for the scripts. (The template's bashio rule is gone:
  # nothing here runs bashio.)
  /tmp/** rwk,

  # The HEALTHCHECK runs curl under this profile against the local Alloy port
  network inet stream,
  network inet6 stream,

  # Access to options.json and the Alloy storage path
  /data/** rw,

  # init-alloy writes the generated config here
  /etc/alloy/ rw,
  /etc/alloy/** rw,

  # init-alloy checks which journal path exists before rendering the config
  /var/log/journal/ r,
  /var/log/journal/** r,
  /run/log/journal/ r,
  /run/log/journal/** r,

  # Start new profile for the Alloy binary
  /usr/bin/alloy cx -> alloy_bin,

  profile alloy_bin flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>
    #include <abstractions/ssl_certs>

    # Receive signals from S6-Overlay
    signal (receive) peer=*_alloy,

    # Push to Loki; serve /-/ready to the HEALTHCHECK's curl and the debug UI
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,

    # The binary and the libraries it dlopens (libsystemd for the journal)
    /usr/bin/alloy mr,
    /usr/lib/** mr,
    /lib/** mr,

    # Generated config, read only
    /etc/alloy/config.alloy r,

    # Storage path (positions, WAL) and the credential files init-alloy
    # writes - not /data/**: options.json holds every secret of the add-on,
    # and Alloy itself needs none of it.
    /data/alloy/ rw,
    /data/alloy/** rwk,
    /data/loki.password r,
    /data/loki.token r,
    /data/metrics.token r,
    /data/remote_write.password r,

    # The host journal, mapped read-only by journald: true
    /var/log/journal/ r,
    /var/log/journal/** r,
    /run/log/journal/ r,
    /run/log/journal/** r,

    # Alloy >= 1.19 (first seen by the CI gate on 1.19.2, denied under 1.13.2's
    # profile): the bundled Snowflake driver extracts its native library to /tmp
    # at startup and mmaps it. It is Alloy's own embedded code, so mapping it is
    # consistent with running the binary at all. Restricted to that path shape.
    owner /tmp/gosnowflake-cgo*/ rw,
    owner /tmp/gosnowflake-cgo*/** rwm,
    # 1.19's usage reporter identifies the host by DMI product UUID. Reporting is
    # off (--disable-reporting); the read is denied QUIETLY rather than logged,
    # so a profile audit stays about things that matter.
    deny /sys/devices/virtual/dmi/id/product_uuid r,

    # What the Go runtime and the journal reader look at
    /etc/machine-id r,
    /proc/sys/net/core/somaxconn r,
    /proc/sys/kernel/hostname r,
    /sys/kernel/mm/transparent_hugepage/hpage_pmd_size r,
    /sys/fs/cgroup/** r,
    owner /proc/*/** r,
    /dev/null rw,
    /tmp/** rwk,

    # Alloy's own resource collector (gopsutil, always registered) runs each
    # time Alloy's metrics are scraped: self_metrics, or anything scraping
    # :12345/metrics. Its one-off container probe reads /proc/modules and PID
    # 1's environment (a ptrace read of the outer profile); it ignores both
    # errors and finds "docker guest" through /.dockerenv anyway, so they are
    # denied quietly - PID 1's environment is no business of Alloy's.
    # In a container it then takes the boot time from /proc/uptime, on every
    # scrape (alloy_resources_process_start_time_seconds).
    /proc/uptime r,
    deny /proc/modules r,
    deny ptrace (read) peer=*_alloy,
  }
}
